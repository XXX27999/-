function UGCGameMode:OnPlayerJoined(PlayerController)
-- 获取玩家Pawn
local Pawn = PlayerController:GetPawn()
if Pawn then
-- 使用V2 API添加默认装备
UGCBackpackSystemV2.AddItemV2(Pawn, WeaponID, 1)
end
end
