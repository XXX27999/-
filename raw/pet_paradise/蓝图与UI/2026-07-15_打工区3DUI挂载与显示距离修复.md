# 打工区 3DUI 挂载与显示距离修复

> 日期：2026-07-15  
> 项目：pet_paradise  
> 范围：所有 WorkingArea 本体挂载统一 Work 3DUI、文本绑定、布局居中、显示距离、闪烁/近距离不显示修复

---

## 1. 功能概述

为全部打工区（`WorkingArea` / `WorkingArea2` / `WorkingArea_2~_11`）在 **Actor 本体** 上挂载统一 3DUI：

- UI 资源：`Asset/Blueprint/Prefabs/UI/districtDisplay/Work.Work_C`
- 显示内容：
  - 基础打工经验 ← `BaseWorkExp`
  - 基础打工金币 ← `BaseWorkGold`
  - 打工区等级 ← `WorkingLevel`（优先 GameAttribute，失败回退字段）
- 常驻挂载，高度偏移 `Z=1000`
- 显示距离 `3000`（30 米），由 `ObjectPositionWidget.ParamInfo.MaxShowDistance` 控制
- 等级 / 金币 / 经验文本与 `UTRichTextBlock_34` 同轴水平居中

旧 `districtDisplay/Works/Work2~12` 标记物挂载链路已停用，避免双 UI。

---

## 2. 已实现功能

- 所有打工区 BeginPlay 注册 3DUI，EndPlay 卸载
- 统一 UI 路径与注册表命名空间 `WorkArea_X`，避免与宠物系统数字 ID 冲突
- 仅本地有 `PlayerController` 的端挂载，避免纯服务器双挂
- Actor 级防重复挂载
- 常驻显示（`AlwaysVisible`），`HideGroup/HideAll` 不会关掉
- UI 文本通过 `Event_InitParamEnd` + `SetText` 刷新（不依赖 `BindingProperty`）
- Tick 每秒补刷文本；**不做** 手动 `SetVisibility`
- 显示距离写入 `ParamInfo.MaxShowDistance=3000`
- UMG 布局：三个 TextBlock 水平中心锚点 + 文本居中 + AutoSize
- 旧标记物脚本禁用 SuperClass，避免蓝图图二次挂载
- 宠物系统不再占用 `WorkDistrictDisplayRegistry`

### 打工区注册映射

| 脚本 | entryKey / workId | group |
|------|-------------------|-------|
| WorkingArea | WorkArea_1 / 1 | 1 |
| WorkingArea_2 | WorkArea_2 / 2 | 1 |
| WorkingArea_3~8 | WorkArea_3~8 | 1 |
| WorkingArea2 | WorkArea_WorkingArea2 | 1 |
| WorkingArea_9~11 | WorkArea_9~11 | 2 |

```lua
-- 脚本: Work.lua（Prefabs/UI/districtDisplay）
---@class Work_C:ObjectPositionWidget
---@field TextBlock_101 UTextBlock
---@field TextBlock_227 UTextBlock
---@field TextBlock_228 UTextBlock
---@field UTRichTextBlock_34 UUTRichTextBlock
--Edit Below--
local Work_C = { bInitDoOnce = false, TargetWorkArea = nil }

--- 打工区 3DUI 最大显示距离（厘米）
--- 必须写到 ParamInfo.MaxShowDistance，这是 ObjectPositionWidget 真正的显示距离字段
local WORK_UI_MAX_SHOW_DISTANCE = 3000

--- 安全设置引擎侧显示距离参数
--- @param self Work_C
local function ApplyShowDistanceParams(self)
	ugcprint("【Work+ApplyShowDistanceParams】=== 开始[设置显示距离] ===")
	ugcprint("【Work+ApplyShowDistanceParams】数据: targetDistance=" .. tostring(WORK_UI_MAX_SHOW_DISTANCE)
		.. ", self=" .. tostring(self))

	local ok, err = pcall(function()
		-- ObjectPositionWidget 官方参数：MaxShowDistance 控制超出距离后隐藏
		if self.ParamInfo then
			self.ParamInfo.MaxShowDistance = WORK_UI_MAX_SHOW_DISTANCE
			-- 当前工程值：BBeOcclusionHide 由业务最终配置决定（见经验章节）
			self.ParamInfo.BBeOcclusionHide = true
			-- 关闭出屏隐藏，避免镜头轻微偏转就整组消失（由引擎 MaxShowDistance 管距离）
			self.ParamInfo.BOutScreenHide = false
			ugcprint("【Work+ApplyShowDistanceParams】分支: 已写 ParamInfo.MaxShowDistance="
				.. tostring(self.ParamInfo.MaxShowDistance)
				.. ", BBeOcclusionHide=" .. tostring(self.ParamInfo.BBeOcclusionHide)
				.. ", BOutScreenHide=" .. tostring(self.ParamInfo.BOutScreenHide))
		else
			ugcprint("【Work+ApplyShowDistanceParams】分支: ParamInfo 为空")
		end

		-- DistanceThreshold 仅用于跨阈值事件，保持与显示距离一致，避免误导
		self.DistanceThreshold = WORK_UI_MAX_SHOW_DISTANCE
		ugcprint("【Work+ApplyShowDistanceParams】分支: DistanceThreshold="
			.. tostring(self.DistanceThreshold))
	end)

	if not ok then
		ugcprint("【Work+ApplyShowDistanceParams】错误: " .. tostring(err))
		ugcprint("【Work+ApplyShowDistanceParams】=== [设置显示距离]完毕：失败 ===")
		return false
	end

	ugcprint("【Work+ApplyShowDistanceParams】=== [设置显示距离]完毕：成功 ===")
	return true
end

--- 从目标打工区读取并刷新三个文本控件
--- @param self Work_C
local function RefreshWorkTexts(self)
	ugcprint("【Work+RefreshWorkTexts】=== 开始[刷新打工区文本] ===")
	local targetActor = self.TargetWorkArea
	if not targetActor then
		local okGet, got = pcall(function()
			return self:GetTargetActor()
		end)
		if okGet then
			targetActor = got
			self.TargetWorkArea = got
		end
	end

	ugcprint("【Work+RefreshWorkTexts】数据: targetActor=" .. tostring(targetActor)
		.. ", DistanceToPlayer=" .. tostring(self.DistanceToPlayer)
		.. ", MaxShowDistance=" .. tostring(self.ParamInfo and self.ParamInfo.MaxShowDistance)
		.. ", WidgetVisible=" .. tostring(self.WidgetVisible))

	if not targetActor then
		ugcprint("【Work+RefreshWorkTexts】分支: targetActor 为空，写入默认文本")
		if self.TextBlock_228 then
			self.TextBlock_228:SetText("基础打工经验：0")
		end
		if self.TextBlock_227 then
			self.TextBlock_227:SetText("基础打工金币：0")
		end
		if self.TextBlock_101 then
			self.TextBlock_101:SetText("打工区等级：0")
		end
		ugcprint("【Work+RefreshWorkTexts】=== [刷新打工区文本]完毕：target为空 ===")
		return
	end

	local exp = targetActor.BaseWorkExp or 0
	local gold = targetActor.BaseWorkGold or 0
	local level = targetActor.WorkingLevel or 0
	local okLevel, levelResult = pcall(function()
		return UGCAttributeSystem.GetGameAttributeValue(targetActor, "WorkingLevel")
	end)
	if okLevel and levelResult ~= nil then
		level = levelResult
		ugcprint("【Work+RefreshWorkTexts】分支: 成功读取 WorkingLevel 属性=" .. tostring(level))
	else
		level = targetActor.WorkingLevel or targetActor.Level or 0
		ugcprint("【Work+RefreshWorkTexts】分支: WorkingLevel 属性读取失败，回退字段 level=" .. tostring(level)
			.. ", err=" .. tostring(levelResult))
	end

	ugcprint("【Work+RefreshWorkTexts】数据: BaseWorkExp=" .. tostring(exp)
		.. ", BaseWorkGold=" .. tostring(gold)
		.. ", WorkingLevel=" .. tostring(level))

	if self.TextBlock_228 then
		self.TextBlock_228:SetText("基础打工经验：" .. tostring(exp))
		ugcprint("【Work+RefreshWorkTexts】分支: 已设置经验文本")
	else
		ugcprint("【Work+RefreshWorkTexts】分支: TextBlock_228 为空")
	end

	if self.TextBlock_227 then
		self.TextBlock_227:SetText("基础打工金币：" .. tostring(gold))
		ugcprint("【Work+RefreshWorkTexts】分支: 已设置金币文本")
	else
		ugcprint("【Work+RefreshWorkTexts】分支: TextBlock_227 为空")
	end

	if self.TextBlock_101 then
		self.TextBlock_101:SetText("打工区等级：" .. tostring(level))
		ugcprint("【Work+RefreshWorkTexts】分支: 已设置等级文本")
	else
		ugcprint("【Work+RefreshWorkTexts】分支: TextBlock_101 为空")
	end

	ugcprint("【Work+RefreshWorkTexts】=== [刷新打工区文本]完毕：成功 ===")
end

function Work_C:Construct()
	ugcprint("【Work+Construct】=== 开始[Work 3DUI Construct] ===")
	ugcprint("【Work+Construct】数据: self=" .. tostring(self)
		.. ", 目标显示距离=" .. tostring(WORK_UI_MAX_SHOW_DISTANCE))

	local ok, err = pcall(function()
		ApplyShowDistanceParams(self)
		self:SetRenderTransformPivot({ X = 0.5, Y = 0.5 })
	end)
	if ok then
		ugcprint("【Work+Construct】分支: Construct 参数设置成功")
	else
		ugcprint("【Work+Construct】错误: " .. tostring(err))
	end
	ugcprint("【Work+Construct】=== [Work 3DUI Construct]完毕 ===")
end

--- ObjectPositionWidget 初始化完成回调：绑定目标打工区并刷新文本
--- 注意：不要在这里/Tick 里手动 SetVisibility，会与引擎 Event_UpdateVisibility 抢控制权，
--- 导致“镜头转过来闪一下全显示然后消失”或“靠近了也不显示”
function Work_C:Event_InitParamEnd()
	ugcprint("【Work+Event_InitParamEnd】=== 开始[初始化打工区3DUI文本] ===")
	ugcprint("【Work+Event_InitParamEnd】数据: self=" .. tostring(self))

	local ok, err = pcall(function()
		-- 再次确保显示距离写入（InitParam 后 ParamInfo 可能被引擎刷新）
		ApplyShowDistanceParams(self)

		local targetActor = self:GetTargetActor()
		self.TargetWorkArea = targetActor
		ugcprint("【Work+Event_InitParamEnd】数据: targetActor=" .. tostring(targetActor)
			.. ", MaxShowDistance=" .. tostring(self.ParamInfo and self.ParamInfo.MaxShowDistance)
			.. ", DistanceToPlayer=" .. tostring(self.DistanceToPlayer))
		RefreshWorkTexts(self)
	end)

	if not ok then
		ugcprint("【Work+Event_InitParamEnd】错误: " .. tostring(err))
	else
		ugcprint("【Work+Event_InitParamEnd】分支: 初始化成功")
	end
	ugcprint("【Work+Event_InitParamEnd】=== [初始化打工区3DUI文本]完毕 ===")
end

--- 仅做低频文本补刷，不做距离显隐（显隐交给引擎 MaxShowDistance）
function Work_C:Tick(MyGeometry, InDeltaTime)
	if not self._WorkTextTickAccum then
		self._WorkTextTickAccum = 0
	end
	self._WorkTextTickAccum = self._WorkTextTickAccum + (InDeltaTime or 0)
	if self._WorkTextTickAccum < 1.0 then
		return
	end
	self._WorkTextTickAccum = 0

	local ok, err = pcall(function()
		if not self.TargetWorkArea then
			self.TargetWorkArea = self:GetTargetActor()
		end
		if self.TargetWorkArea then
			RefreshWorkTexts(self)
		else
			ugcprint("【Work+Tick】分支: TargetWorkArea 仍为空，等待下次")
		end
	end)
	if not ok then
		ugcprint("【Work+Tick】错误: " .. tostring(err))
	end
end

function Work_C:Destruct()
	ugcprint("【Work+Destruct】=== 开始[销毁Work 3DUI] ===")
	self.TargetWorkArea = nil
	ugcprint("【Work+Destruct】=== [销毁Work 3DUI]完毕 ===")
end

-- [Editor Generated Lua] function define Begin:
function Work_C:LuaInit()
	if self.bInitDoOnce then
		return;
	end
	self.bInitDoOnce = true;
	-- BindingProperty 对 ObjectPositionWidget 不可靠，文本改由 Event_InitParamEnd/Tick 直接 SetText
	ugcprint("【Work+LuaInit】=== 完成[跳过BindingProperty，改用SetText] ===")
end

-- [Editor Generated Lua] function define End;

return Work_C
```

### 3.2 挂载工具：`WorkingAreaDisplayMount.lua`

路径：`Script/Common/Utils/WorkingAreaDisplayMount.lua`

```lua
-- 脚本: WorkingAreaDisplayMount.lua
--- WorkingArea 3DUI 挂载工具：将统一 Work 3DUI 注册到 WorkDistrictDisplayRegistry
local WorkDistrictDisplayRegistry = require("Script.Common.Utils.WorkDistrictDisplayRegistry")

local WorkingAreaDisplayMount = {}

--- 统一打工区 3DUI 资源路径（挂载到所有 WorkingArea）
local DEFAULT_UI_PATH = "Asset/Blueprint/Prefabs/UI/districtDisplay/Work.Work_C"
--- 默认头顶偏移（常驻显示高度）
local DEFAULT_OFFSET = { X = 0, Y = 0, Z = 1000 }
--- 打工区 UI 是否常驻显示（不受 WorkTrigger 隐藏）
local ALWAYS_VISIBLE = true
--- 注册表命名空间前缀，避免与宠物等其它系统共用数字 workId 冲突
local ENTRY_KEY_PREFIX = "WorkArea_"

--- 生成注册表唯一 key
--- @param workId number|string
--- @return string
local function MakeEntryKey(workId)
    return ENTRY_KEY_PREFIX .. tostring(workId)
end

--- 按 workId 推断显示分组：2-8 与 1 属 group1，9-12 属 group2
--- @param workId number|string 打工区编号
--- @return number groupId
local function ResolveDefaultGroupId(workId)
    local numericId = tonumber(workId) or 0
    if numericId >= 9 then
        return 2
    end
    return 1
end

--- 是否应在本端挂载 3DUI：仅本地客户端/主机可见端挂载，避免双端各挂一份
--- @param actor AActor
--- @return boolean
local function ShouldMountOnThisSide(actor)
    if not actor then
        return false
    end

    -- 纯服务器无本地玩家，不挂载
    local localPC = UGCGameSystem.GetLocalPlayerController()
    if not localPC then
        ugcprint("【WorkingAreaDisplayMount+ShouldMountOnThisSide】分支: 无本地PlayerController，跳过挂载")
        return false
    end

    return true
end

--- 注册打工区 3DUI（BeginPlay 调用）
--- @param actor AActor 打工区 Actor
--- @param workId number|string 打工区唯一 ID
--- @param groupId number|nil 显示分组（不传则按 workId 推断）
--- @param uiPath string|nil 可选自定义 UI 路径
--- @param offset table|nil 可选偏移
function WorkingAreaDisplayMount.Register(actor, workId, groupId, uiPath, offset)
    ugcprint("【WorkingAreaDisplayMount+Register】=== 开始[注册打工区3DUI] ===")
    ugcprint("【WorkingAreaDisplayMount+Register】数据: actor=" .. tostring(actor)
        .. ", workId=" .. tostring(workId)
        .. ", groupId=" .. tostring(groupId)
        .. ", uiPath=" .. tostring(uiPath)
        .. ", HasAuthority=" .. tostring(actor and actor.HasAuthority and actor:HasAuthority()))

    if not actor or workId == nil then
        ugcprint("【WorkingAreaDisplayMount+Register】错误: actor 或 workId 为空")
        ugcprint("【WorkingAreaDisplayMount+Register】=== [注册打工区3DUI]完毕：失败 ===")
        return false
    end

    if not ShouldMountOnThisSide(actor) then
        ugcprint("【WorkingAreaDisplayMount+Register】分支: 当前端不挂载，跳过")
        ugcprint("【WorkingAreaDisplayMount+Register】=== [注册打工区3DUI]完毕：跳过 ===")
        return false
    end

    -- 同一 Actor 防重复挂载（BeginPlay 重复触发/热更重入）
    if actor.WorkDistrictUIMounted == true and actor.WorkDistrictUIEntryKey then
        ugcprint("【WorkingAreaDisplayMount+Register】分支: Actor 已挂载过，跳过重复注册 entryKey="
            .. tostring(actor.WorkDistrictUIEntryKey))
        ugcprint("【WorkingAreaDisplayMount+Register】=== [注册打工区3DUI]完毕：跳过重复 ===")
        return true
    end

    local entryKey = MakeEntryKey(workId)
    local resolvedGroupId = groupId or ResolveDefaultGroupId(workId)
    local resolvedUiPath = uiPath or DEFAULT_UI_PATH
    local resolvedOffset = offset or DEFAULT_OFFSET

    ugcprint("【WorkingAreaDisplayMount+Register】数据: entryKey=" .. tostring(entryKey)
        .. ", resolvedOffset.Z=" .. tostring(resolvedOffset.Z)
        .. ", alwaysVisible=" .. tostring(ALWAYS_VISIBLE))

    local ok, err = pcall(function()
        WorkDistrictDisplayRegistry.Register(
            entryKey,
            resolvedGroupId,
            actor,
            resolvedUiPath,
            resolvedOffset,
            ALWAYS_VISIBLE
        )
    end)

    if ok then
        actor.WorkDistrictUIMounted = true
        actor.WorkDistrictUIEntryKey = entryKey
        ugcprint("【WorkingAreaDisplayMount+Register】分支: 注册成功 entryKey=" .. tostring(entryKey)
            .. ", groupId=" .. tostring(resolvedGroupId)
            .. ", uiPath=" .. tostring(resolvedUiPath)
            .. ", offsetZ=" .. tostring(resolvedOffset.Z)
            .. ", alwaysVisible=" .. tostring(ALWAYS_VISIBLE))
        ugcprint("【WorkingAreaDisplayMount+Register】=== [注册打工区3DUI]完毕：成功 ===")
        return true
    end

    ugcprint("【WorkingAreaDisplayMount+Register】错误: " .. tostring(err))
    ugcprint("【WorkingAreaDisplayMount+Register】=== [注册打工区3DUI]完毕：失败 ===")
    return false
end

--- 卸载打工区 3DUI（EndPlay 调用）
--- @param actor AActor 打工区 Actor
--- @param workId number|string 打工区唯一 ID
function WorkingAreaDisplayMount.Unregister(actor, workId)
    ugcprint("【WorkingAreaDisplayMount+Unregister】=== 开始[卸载打工区3DUI] ===")
    ugcprint("【WorkingAreaDisplayMount+Unregister】数据: actor=" .. tostring(actor) .. ", workId=" .. tostring(workId))

    if workId == nil then
        ugcprint("【WorkingAreaDisplayMount+Unregister】错误: workId 为空")
        ugcprint("【WorkingAreaDisplayMount+Unregister】=== [卸载打工区3DUI]完毕：失败 ===")
        return false
    end

    local entryKey = (actor and actor.WorkDistrictUIEntryKey) or MakeEntryKey(workId)
    local ok, err = pcall(function()
        WorkDistrictDisplayRegistry.Unregister(entryKey, actor)
    end)

    if actor then
        actor.WorkDistrictUIMounted = false
        actor.WorkDistrictUIEntryKey = nil
    end

    if ok then
        ugcprint("【WorkingAreaDisplayMount+Unregister】分支: 卸载成功 entryKey=" .. tostring(entryKey))
        ugcprint("【WorkingAreaDisplayMount+Unregister】=== [卸载打工区3DUI]完毕：成功 ===")
        return true
    end

    ugcprint("【WorkingAreaDisplayMount+Unregister】错误: " .. tostring(err))
    ugcprint("【WorkingAreaDisplayMount+Unregister】=== [卸载打工区3DUI]完毕：失败 ===")
    return false
end

return WorkingAreaDisplayMount
```

### 3.3 注册表：`WorkDistrictDisplayRegistry.lua`

```lua
-- 脚本: WorkDistrictDisplayRegistry.lua
--- 打工区 3DUI 显示注册表：按 group 控制显示/隐藏，真正挂载目标为 WorkingArea Actor
local WorkDistrictDisplayRegistry = {
    Entries = {},
    VisibleGroups = {}
}

local DEFAULT_OFFSET = { X = 0, Y = 0, Z = 0 }

--- 校验 Actor 是否有效
--- @param actor AActor|nil
--- @return boolean
local function IsValidActor(actor)
    if not actor then
        return false
    end

    if UE and UE.IsValid then
        local ok, valid = pcall(function()
            return UE.IsValid(actor)
        end)
        if ok then
            return valid == true
        end
    end

    return true
end

--- 显示单条注册项：调用 AddObjectPositionUI 挂载 3DUI
--- @param entry table
local function ShowEntry(entry)
    ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】=== 开始[显示打工区3DUI] ===")
    ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】数据: entry=" .. tostring(entry)
        .. ", Actor=" .. tostring(entry and entry.Actor)
        .. ", UIPath=" .. tostring(entry and entry.UIPath)
        .. ", InstanceIndex=" .. tostring(entry and entry.InstanceIndex)
        .. ", offsetZ=" .. tostring(entry and entry.Offset and entry.Offset.Z))

    if not entry or entry.InstanceIndex ~= nil or not IsValidActor(entry.Actor) or not entry.UIPath then
        ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】分支: 跳过显示（已显示/无效/无路径）")
        ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】=== [显示打工区3DUI]完毕：跳过 ===")
        return
    end

    -- 纯服务器无本地玩家时不创建 UI，避免双端各挂一份
    local localPC = UGCGameSystem.GetLocalPlayerController()
    if not localPC then
        ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】分支: 无本地PlayerController，跳过 AddObjectPositionUI")
        ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】=== [显示打工区3DUI]完毕：跳过 ===")
        return
    end

    local ok, err = pcall(function()
        local fullPath = UGCGameSystem.GetUGCResourcesFullPath(entry.UIPath)
        ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】数据: fullPath=" .. tostring(fullPath))
        entry.InstanceIndex = UGCWidgetManagerSystem.AddObjectPositionUI(
            entry.Actor,
            fullPath,
            entry.Offset or DEFAULT_OFFSET,
            true,
            true,
            false,
            true
        )
        ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】分支: AddObjectPositionUI 完成 InstanceIndex="
            .. tostring(entry.InstanceIndex))
    end)

    if not ok then
        ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】错误: " .. tostring(err))
        entry.InstanceIndex = nil
        ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】=== [显示打工区3DUI]完毕：失败 ===")
        return
    end

    ugcprint("【WorkDistrictDisplayRegistry+ShowEntry】=== [显示打工区3DUI]完毕：成功 ===")
end

--- 隐藏单条注册项：调用 RemoveObjectPositionUI
--- @param entry table
local function HideEntry(entry)
    ugcprint("【WorkDistrictDisplayRegistry+HideEntry】=== 开始[隐藏打工区3DUI] ===")
    ugcprint("【WorkDistrictDisplayRegistry+HideEntry】数据: entry=" .. tostring(entry)
        .. ", Actor=" .. tostring(entry and entry.Actor)
        .. ", InstanceIndex=" .. tostring(entry and entry.InstanceIndex))

    if not entry or entry.InstanceIndex == nil or not IsValidActor(entry.Actor) then
        if entry then
            entry.InstanceIndex = nil
        end
        ugcprint("【WorkDistrictDisplayRegistry+HideEntry】分支: 无需移除或 Actor 无效")
        ugcprint("【WorkDistrictDisplayRegistry+HideEntry】=== [隐藏打工区3DUI]完毕：跳过 ===")
        return
    end

    local ok, err = pcall(function()
        UGCWidgetManagerSystem.RemoveObjectPositionUI(entry.Actor, entry.InstanceIndex)
    end)
    entry.InstanceIndex = nil

    if not ok then

--- 注册打工区显示项
--- @param workId number|string
--- @param groupId number
--- @param actor AActor
--- @param uiPath string
--- @param offset table|nil
--- @param alwaysVisible boolean|nil 为 true 时注册后立即常驻显示，且不被 HideGroup 隐藏
function WorkDistrictDisplayRegistry.Register(workId, groupId, actor, uiPath, offset, alwaysVisible)
    ugcprint("【WorkDistrictDisplayRegistry+Register】=== 开始[注册显示项] ===")
    ugcprint("【WorkDistrictDisplayRegistry+Register】数据: workId=" .. tostring(workId)
        .. ", groupId=" .. tostring(groupId)
        .. ", actor=" .. tostring(actor)
        .. ", uiPath=" .. tostring(uiPath)
        .. ", alwaysVisible=" .. tostring(alwaysVisible)
        .. ", offsetZ=" .. tostring(offset and offset.Z))

    if not workId or not groupId or not actor or not uiPath then
        ugcprint("【WorkDistrictDisplayRegistry+Register】错误: 参数不完整")
        ugcprint("【WorkDistrictDisplayRegistry+Register】=== [注册显示项]完毕：失败 ===")
        return
    end

    local oldEntry = WorkDistrictDisplayRegistry.Entries[workId]
    if oldEntry and oldEntry.Actor ~= actor then
        ugcprint("【WorkDistrictDisplayRegistry+Register】分支: workId 已存在且 Actor 不同，先隐藏旧项")
        HideEntry(oldEntry)
    elseif oldEntry and oldEntry.Actor == actor and oldEntry.InstanceIndex ~= nil then
        ugcprint("【WorkDistrictDisplayRegistry+Register】分支: 同一 Actor 已有实例，复用 InstanceIndex="
            .. tostring(oldEntry.InstanceIndex))
    else
        ugcprint("【WorkDistrictDisplayRegistry+Register】分支: 无冲突旧项或同一 Actor 覆盖注册")
    end

    local entry = {
        GroupId = groupId,
        Actor = actor,
        UIPath = uiPath,
        Offset = offset or DEFAULT_OFFSET,
        AlwaysVisible = alwaysVisible == true,
        InstanceIndex = oldEntry and oldEntry.Actor == actor and oldEntry.InstanceIndex or nil
    }
    WorkDistrictDisplayRegistry.Entries[workId] = entry

    if entry.AlwaysVisible then
        ugcprint("【WorkDistrictDisplayRegistry+Register】分支: 常驻显示，立即 ShowEntry")
        WorkDistrictDisplayRegistry.VisibleGroups[groupId] = true
        ShowEntry(entry)
    elseif WorkDistrictDisplayRegistry.VisibleGroups[groupId] == true then
        ugcprint("【WorkDistrictDisplayRegistry+Register】分支: 分组已可见，立即 ShowEntry")
        ShowEntry(entry)
    else
        ugcprint("【WorkDistrictDisplayRegistry+Register】分支: 分组当前不可见，仅注册不显示")
    end

    ugcprint("【WorkDistrictDisplayRegistry+Register】=== [注册显示项]完毕：成功 ===")
end

--- 注销打工区显示项

    HideEntry(entry)
    WorkDistrictDisplayRegistry.Entries[workId] = nil
    ugcprint("【WorkDistrictDisplayRegistry+Unregister】=== [注销显示项]完毕：成功 ===")
end
```

## 6. 关键经验

1. **`ObjectPositionWidget` 显示距离 = `ParamInfo.MaxShowDistance`**  
   - `DistanceThreshold` 只用于跨阈值事件  
   - `bIsBeyondContiousShowDis` 是只读状态位，不是配置项  

2. **禁止对 ObjectPositionWidget 手动 `SetVisibility` 做距离裁剪**  
   - 会与引擎每帧 `Event_UpdateVisibility` 打架  
   - 典型症状：转镜头闪一下、靠近后永久不显示  

3. **ObjectPositionWidget 文本刷新用 `Event_InitParamEnd` + `SetText`**  
   - `BindingProperty("Text", ...)` 在该基类上不可靠  

4. **3DUI 挂载目标必须是业务 Actor 本体**  
   - 标记物与本体分离时，数据源/位置都会错  

5. **注册表 key 必须有命名空间**  
   - 数字 `workId` 容易与宠物等其它系统冲突  

6. **双端防重**  
   - 无本地 `PlayerController` 时不要 `AddObjectPositionUI`  
   - Actor 上记 `WorkDistrictUIMounted` 防 BeginPlay 重入  

7. **UMG 布局**  
   - `widget_slot` 对 Anchors/Position 支持有限  
   - 应直接改 `CanvasPanelSlot.LayoutData` 后 `save_package`  

8. **停用旧逻辑时不要调用 SuperClass.BeginPlay**  
   - 蓝图事件图里可能还有硬编码 `AddObjectPositionUI`  

9. **修改后生效方式**  
   - 涉及 BeginPlay 注册、UI 蓝图布局、ParamInfo 默认值：**需要重新调试 PIE**
